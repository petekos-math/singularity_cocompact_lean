theorem Singularity.cocompact_fuchsian_hittingLaw_singular : ∀
  (Γ : Subgroup (Matrix.ProjectiveSpecialLinearGroup (Fin 2) ℝ)) [DiscreteTopology ↥Γ]
  [CompactSpace (Quotient (MulAction.orbitRel (↥Γ) UpperHalfPlane))] [inst : MeasurableSpace ↥Γ]
  [MeasurableSingletonClass ↥Γ],
  Singularity.ProjectiveNonelementary Γ →
    ∀ (s : Finset ↥Γ) (μ : ↥Γ → ℝ) (hpos : ∀ g ∈ s, 0 < μ g) (hmass : ∑ g ∈ s, μ g = 1),
      Submonoid.closure ↑s = ⊤ →
        ∀ (z : UpperHalfPlane) (b : (ℕ → ↥s) → OnePoint ℝ),
          (∀ᵐ (ω : ℕ → ↥s) ∂Singularity.infiniteWalkLaw s μ ⋯ hmass,
              Filter.Tendsto (fun n => Singularity.hyperbolicCompactEmbedding (Singularity.walkPosition s 1 n ω • z))
                Filter.atTop (nhds (Singularity.compactBoundaryEmbedding (b ω)))) →
            (Singularity.walkBoundaryLaw s μ ⋯ hmass b).MutuallySingular (Singularity.compactPoissonMeasure z) :=
fun Γ [DiscreteTopology ↥Γ] [CompactSpace (Quotient (MulAction.orbitRel (↥Γ) UpperHalfPlane))] [MeasurableSpace ↥Γ]
    [MeasurableSingletonClass ↥Γ] hne s μ hpos hmass hgen z b hb =>
  have he :=
    MeasureTheory.Measure.map_congr
      (Filter.mp_mem (Singularity.cocompact_fuchsian_boundaryMap_tendsto Γ hne s μ hpos hmass hgen z)
        (Filter.mp_mem hb
          (Filter.univ_mem'
            (id fun ω h₁ h₂ => Singularity.compactBoundaryEmbedding_injective (tendsto_nhds_unique h₁ h₂)))));
  Eq.mpr (id (congrArg (fun _a => _a.MutuallySingular (Singularity.compactPoissonMeasure z)) he))
    (Singularity.cocompact_fuchsian_hittingMeasure_singular Γ hne s μ hpos hmass hgen z)
